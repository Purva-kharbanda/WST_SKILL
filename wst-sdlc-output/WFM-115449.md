# Triage: WFM-115449 — Employee Attribute Type/Sub-Type blank; EMP_SENIORITY_DATE default incorrect

```sql
-- Diagnostic (run on repro environment — QA06 TESTING1 or customer SB)
SELECT NAME, DATA_TYPE, DEFAULT_VALUE, DISPLAY_TYPE,
       CONTROL_TYPE, CONTROL_SUB_TYPE, DISP_SEQ_NO
FROM RWS_ATTR
WHERE OWNER_ID = $ownerId
  AND CLIENT_ID = '$clientId'
  AND ENTITY_TYPE = 'T'
  AND NAME IN ('IS_FLOATER', 'EMP_SENIORITY_DATE');

-- Find all legacy control types not supported on v45 Employee Attribute UI
SELECT NAME, CONTROL_TYPE, CONTROL_SUB_TYPE
FROM RWS_ATTR
WHERE OWNER_ID = $ownerId
  AND CLIENT_ID = '$clientId'
  AND ENTITY_TYPE = 'T'
  AND (CONTROL_TYPE NOT IN ('B', 'G') OR CONTROL_SUB_TYPE NOT IN ('B', 'G'));

-- WFM-8487 pattern (IS_FLOATER) — VERIFY owner/client; extend to other rows as needed
UPDATE RWS_ATTR
SET CONTROL_TYPE = 'G', CONTROL_SUB_TYPE = 'G'
WHERE OWNER_ID = $ownerId
  AND CLIENT_ID = '$clientId'
  AND ENTITY_TYPE = 'T'
  AND NAME = 'IS_FLOATER';
```
